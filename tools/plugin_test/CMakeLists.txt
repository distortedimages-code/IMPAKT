# Console test that drives ImpaktProcessor directly (no GUI): state round-trip, host sync, MIDI.
cmake_minimum_required(VERSION 3.22)
project(IMPAKT_TEST VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(${JUCE_PATH} ${CMAKE_BINARY_DIR}/JUCE)
juce_add_console_app(impakt_test PRODUCT_NAME "impakt_test" NEEDS_WEB_BROWSER TRUE)
juce_add_binary_data(IMPAKT_UI_T SOURCES ../../plugin/Source/ui/index.html HEADER_NAME BinaryData.h NAMESPACE BinaryData)
file(GLOB_RECURSE IMPAKT_MI_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../../core/mi/plaits/*.cc ${CMAKE_CURRENT_SOURCE_DIR}/../../core/mi/beads/*.cc ${CMAKE_CURRENT_SOURCE_DIR}/../../core/mi/warps/*.cc ${CMAKE_CURRENT_SOURCE_DIR}/../../core/mi/stmlib/*.cc)
list(FILTER IMPAKT_MI_SOURCES EXCLUDE REGEX ".*/atan\\.cc$")
set(IMPAKT_CORE ../../core/sk_engine.cpp ../../core/sk_kit.cpp ../../core/sk_mod.cpp ../../core/sk_glitch.cpp ../../core/sk_halo.cpp ../../core/sk_strum.cpp ../../core/sk_choir.cpp ../../core/sk_alloy.cpp ../../core/sk_chimera.cpp ../../core/sk_cube.cpp ../../core/sk_west.cpp ../../core/sk_howl.cpp ../../core/sk_stencil.cpp ../../core/sk_mantra.cpp ../../core/sk_radio.cpp ../../core/sk_radio_dec.cpp ../../core/sk_spring.cpp ../../core/sk_weft.cpp ../../core/sk_bank.cpp ../../core/sk_break.cpp ../../core/sk_fx.cpp ../../core/sk_fuzz.cpp ../../core/sk_send.cpp ../../core/sk_box.cpp ../../core/sk_spec.cpp ../../core/sk_org.cpp ../../core/sk_master.cpp ../../core/sk_rev.cpp ../../core/sk_vintage.cpp ../../core/sk_mix.cpp ../../core/sk_macro.cpp ../../core/sk_grain.cpp ../../core/sk_ott.cpp ../../core/sk_lfo_tables.cpp ../../core/sk_pat.cpp ${IMPAKT_MI_SOURCES})
target_sources(impakt_test PRIVATE main.cpp ../../plugin/Source/PluginProcessor.cpp ../../plugin/Source/PluginEditor.cpp ../../plugin/Source/RadioNet.cpp ../../plugin/Source/Updater.cpp ${IMPAKT_CORE})
target_include_directories(impakt_test PRIVATE ../../core ../../core/mi ../../plugin/Source)
target_compile_definitions(impakt_test PRIVATE JUCE_WEB_BROWSER=1 JUCE_USE_CURL=0 JucePlugin_Name="IMPAKT" JucePlugin_VersionString="0.0.0")
target_link_libraries(impakt_test PRIVATE IMPAKT_UI_T juce::juce_audio_utils juce::juce_gui_extra juce::juce_recommended_config_flags)
